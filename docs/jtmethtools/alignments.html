 35def alignment_overlaps_region(
 36        alignment: AlignedSegment,
 37        regions: Regions) -> bool | str:
 38    """Check if an alignment overlaps a region. Returns False if not,
 39    and the name of the first overlapping region if it does.
 40
 41    samtools is much faster if you just want to filter a large BAM."""
 42    ref = alignment.reference_name
 43
 44    try:
 45        regStart, regEnd = regions.starts_ends_of_chrm(ref)
 46    except KeyError:
 47        logger.trace('Reference contig not found: ' + str(alignment.reference_name))
 48        return False
 49
 50    m = (
 51            (alignment.reference_start < regEnd)
 52            & (alignment.reference_end > regStart)
 53    )
 54
 55    isoverlap = np.any(m)
 56
 57    if isoverlap:
 58        reg_name = regions.names[ref][m][0]
 59        logger.trace(reg_name)
 60        return reg_name #its fine
 61    return False

 64def get_bismark_met_str(a: AlignedSegment) -> str|None:
 65    """Return the methylation string for a Bismark alignment, or None if it can't be found."""
 66    # this works in every case I've seen...
 67    tags = a.get_tags()
 68    tag = None
 69    if len(tags) > 2:
 70        tag, met = a.get_tags()[2]
 71    try:
 72        assert tag == 'XM'
 73    # but it's not guaranteed that the tag position will never change
 74    #   so this should work whever it is.
 75    except AssertionError:
 76        for t, m in tags:
 77            if t == 'XM':
 78                tag, met = t, m
 79                break
 80        if tag != 'XM':
 81            logger.warning(f"Can't find XM tag in alignment {a.query_name}, {tags=}")
 82            met = None
 83
 84    return met

 86def write_bam_from_pysam(
 87        out_fn:str,
 88        alignments:Collection[pysam.AlignedSegment],
 89        header_file:str=None,
 90        header:AlignmentHeader=None,
 91):
 92    """Convenience function for writing pysam.AlignedSegments.
 93    Supply a SAM/BAM file name to
 94    use for the header, or supply the header directly. Either
 95    `header_file` or `header`, not both.
 96
 97    Output format (BAM or SAM) determined from out_fn.
 98    """
 99    if (
100           ( (header_file is None) and (header is None) )
101        or ( (header_file is not None) and (header is not None) )
102    ):
103        raise ValueError("One of either `header_file` or `header` must be supplied.")
104
105    if header_file is not None:
106        header = AlignmentFile(header_file).header
107
108    if str(out_fn).lower().endswith('.bam'):
109        mode = 'wb'
110    elif str(out_fn).lower().endswith('.sam'):
111        mode = 'w'
112    else:
113        raise ValueError(f"Can't determine output file type from file name {out_fn}. "
114                         f"Needs to end with .sam or .bam.")
115    with AlignmentFile(out_fn, mode=mode, header=header) as out:
116        for a in alignments:
117            if a is not None:
118                out.write(a)

121def flag_to_text(flag):
122    """
123    Convert the integer flag from pysam AlignedSegment to a human-readable text representation.
124    """
125    # Flag definitions according to the SAM format specification
126    flag_descriptions = {
127        0x1: "Paired read",
128        0x2: "Properly paired",
129        0x4: "Unmapped read",
130        0x8: "Mate unmapped",
131        0x10: "Read reverse strand",
132        0x20: "Mate reverse strand",
133        0x40: "First in pair",
134        0x80: "Second in pair",
135        0x100: "Not primary alignment",
136        0x200: "Read fails platform/vendor quality checks",
137        0x400: "PCR or optical duplicate",
138        0x800: "Supplementary alignment"
139    }
140
141    result = []
142
143    # Check each flag bit and append the description if the bit is set
144    for flag_value, description in flag_descriptions.items():
145        if flag & flag_value:
146            result.append(description)
147
148    return ", ".join(result) if result else "None"

323    @cached_property
324    def _locus_values(self, ) \
325            -> LocusValues:
326        """Get object with attributes "qualities", "nucleotides", and "methylations"
327        mapping each reference locus to the value, reference_locus -> value.
328        """
329
330        empty_return =  LocusValues({}, {}, {}, is_good=False)
331        if not self._has_metstr():
332            logger.warning(
333                f"Can't determine metstr for alignment of {self.a.query_name}, skipping."
334            )
335            count_alignment_error()
336            return empty_return
337
338        for segment in (self.a, self.a2):
339            if segment is None:
340                continue
341
342            align_len = sum([x[1] for x in segment.cigartuples if x[0] in {0, 1, 7, 8}])
343            metstr_len = len(get_bismark_met_str(segment))
344            phred_len = len(segment.query_qualities)
345            nt_len = len(segment.query_sequence)
346            if not (align_len == metstr_len == phred_len == nt_len):
347                logger.warning(
348                    f"Length mismatch of methylation string for alignment of {self.a.query_name}, skipping.\n"
349                    f"({align_len=}, {metstr_len=}, {phred_len=}, {nt_len=} {segment.cigartuples=}, {self.filename=})"
350                )
351                count_alignment_error()
352                return empty_return
353
354        if self.a2 is None:
355            phreds, methylations, nucleotides = {}, {}, {}
356            for q_pos, r_pos in self.a.get_aligned_pairs(matches_only=True):
357                phreds[r_pos] = self.a.query_qualities[q_pos]
358                nucleotides[r_pos] = self.a.query_sequence[q_pos]
359                methylations[r_pos] = self._get_met_str(self.a)[q_pos]
360        else: # it's a paired alignment
361            if self.use_quality_profile:
362                from jtmethtools._quality_profiles import quality_profile_match_41, quality_profile_mismatch_41
363            else:
364                quality_profile_match_41, quality_profile_mismatch_41 = None, None
365
366            # .aligned_pairs is reference position->read locus
367            a1_loc_pos, a2_loc_pos = [
368                {r: q for (q, r) in ap if (q is not None) and (r is not None)}
369                for ap in (self.a.get_aligned_pairs(), self.a2.get_aligned_pairs())
370            ]
371            # a1_loc_pos, a2_loc_pos = [
372            #     dict(aln.get_aligned_pairs(matches_only=True))
373            #     for aln in (self.a, self.a2)
374            # ]
375
376            a1_loc = set(a1_loc_pos.keys())
377            a2_loc = set(a2_loc_pos.keys())
378            shared_loc = a1_loc.intersection(a2_loc)
379            a1_only = a1_loc.difference(a2_loc)
380            a2_only = a2_loc.difference(a1_loc)
381
382            phreds = {}
383            nucleotides = {}
384            methylations = {}
385
386            a1_metstr = get_bismark_met_str(self.a)
387            a2_metstr = get_bismark_met_str(self.a2)
388
389            try:
390                # get the values where the position only exists in one of the mates
391                for only_loc, loc_pos, a, metstr in (
392                        (a1_only, a1_loc_pos, self.a, a1_metstr),
393                        (a2_only, a2_loc_pos, self.a2, a2_metstr)
394                ):
395                    for l in only_loc:
396                        p = loc_pos[l]
397                        phreds[l] = a.query_qualities[p]
398                        nucleotides[l] = a.query_sequence[p]
399                        methylations[l] = metstr[p]
400            except:
401                print('BAM = ', self.filename, ' | Read = ', self.a.query_name)
402                raise
403
404            if (not self.prefer_R1) or self.use_quality_profile:
405                for l in shared_loc:
406                    a1_pos = a1_loc_pos[l]
407                    a2_pos = a2_loc_pos[l]
408
409                    a1_nt = self.a.query_sequence[a1_pos]
410                    a2_nt = self.a2.query_sequence[a2_pos]
411
412                    a1_phred = self.a.query_qualities[a1_pos]
413                    a2_phred = self.a2.query_qualities[a2_pos]
414
415                    a1_met = a1_metstr[a1_pos]
416                    a2_met = a2_metstr[a2_pos]
417
418                    # Keep the nucleotide with the highest quality
419                    # (in the case that a1 and a2 have different nucleotides and
420                    #   the phred is the same, we'll keep the a1 NT)
421                    if a1_phred >= a2_phred:
422                        nt = a1_nt
423                        met = a1_met
424                    else:
425                        nt = a2_nt
426                        met = a2_met
427
428                    if self.use_quality_profile:
429                        if a1_nt == a2_nt:
430                            phred = quality_profile_match_41[a1_phred][a2_phred]
431                        else:
432                            phred = quality_profile_mismatch_41[a1_phred][a2_phred]
433                    else:
434                        phred = max((a1_phred, a2_phred))
435
436                    phreds[l] = phred
437                    nucleotides[l] = nt
438                    methylations[l] = met
439            else:
440                # use R1
441                for l in shared_loc:
442                    a1, loc_pos, metstr = (
443                        (self.a, a1_loc_pos, a1_metstr)
444                        if self.a.is_read1
445                        else (self.a2, a2_loc_pos, a2_metstr)
446                    )
447                    a1_pos = loc_pos[l]
448                    phreds[l] = a1.query_qualities[a1_pos]
449                    nucleotides[l] = a1.query_sequence[a1_pos]
450                    methylations[l] = a1_metstr[a1_pos]
451
452        sorted_phreds, sorted_nucleotides, sorted_methylations = (
453            dict(sorted(d.items()))
454            for d in (phreds, nucleotides, methylations)
455        )
456
457        return LocusValues(qualities=sorted_phreds, nucleotides=sorted_nucleotides, methylations=sorted_methylations)

459    @property
460    def fragment_length(self) -> typing.Union[int, Literal[np.nan]]:
461        """Fragment length from the 5' ends of a read pair (cfDNAPro convention).
462
463        Returns the distance from the 5' end of the forward-strand mate to the
464        5' end of the reverse-strand mate. Unlike the outer span (or BAM TLEN),
465        this stays correct for dovetailed pairs, where a fragment shorter than
466        the read length causes each mate to run past the other's 5' end.
467
468        Returns np.nan if the pair can't yield a meaningful length.
469        """
470        a, b = self.a, self.a2
471        if b is None:
472            return np.nan
473
474        # Both mates must be mapped; reference_end is None for unmapped reads
475        # and for reads whose CIGAR consumes no reference.
476        if a.is_unmapped or b.is_unmapped:
477            return np.nan
478        if a.reference_end is None or b.reference_end is None:
479            return np.nan
480
481        # Same chromosome, opposite orientation (FR or RF).
482        if a.is_reverse == b.is_reverse:
483            return np.nan
484
485        fwd, rev = self.get_fwd_rev()
486
487        # reference_start is 0-based inclusive, reference_end 0-based exclusive,
488        # so the difference is the span directly — no +1.
489        length = rev.reference_end - fwd.reference_start
490
491        # Outward-facing pairs give a negative or degenerate span.
492        if length <= 0:
493            return np.nan
494
495        return length
496
497    @property
498    def fragment_start_end(self) -> tuple[int, int]:
499        """Fragment start end position in reference positions, correcting for readthroughs.
500        Where reads dovetail outwards, the overlapping span is used.
501        See https://doi.org/10.1186/s13059-025-03607-5"""
502
503        fwd, rev = self.get_fwd_rev()
504
505        if fwd.reference_start > rev.reference_start:
506            start = rev.reference_start
507        else:
508            start = min(fwd.reference_start, rev.reference_start)
509
510        if rev.reference_end > fwd.reference_end:
511            end = rev.reference_end
512        else:
513            end = max(fwd.reference_id, rev.reference_end)
514
515        return start, end

529

531        locs = list(self.locus_methylation.keys())

535
536        start = min(locs)
537        for pos in range(start, max(locs) + 1):

540                metstr.append('-')
541            else:
542                metstr.append(m)

612def _iter_bam_pe_qname_sorted(
613        bam:str|Path|AlignmentFile,
614) -> Iterable[Tuple[AlignedSegment, AlignedSegment|None]]:
615    """Iterate over a paired-end bam file, yielding pairs of alignments.
616    Where a read is unpaired, yield (alignment, None).
617    """
618
619    bam = _load_bam(bam)
620
621    if not bam.header.get('HD', {}).get('SO', 'Unknown') == 'queryname':
622        raise RuntimeError(f'BAM file must be sorted by queryname')
623
624    aln_prev: pysam.AlignedSegment | None = None
625    for i, aln_current in enumerate(bam):

462        """Fragment length from the 5' ends of a read pair (cfDNAPro convention).
463
464        Returns the distance from the 5' end of the forward-strand mate to the
465        5' end of the reverse-strand mate. Unlike the outer span (or BAM TLEN),
466        this stays correct for dovetailed pairs, where a fragment shorter than
467        the read length causes each mate to run past the other's 5' end.
468

470        """

472        if b is None:
473            return np.nan
474
475        # Both mates must be mapped; reference_end is None for unmapped reads
476        # and for reads whose CIGAR consumes no reference.
477        if a.is_unmapped or b.is_unmapped:
478            return np.nan
479        if a.reference_end is None or b.reference_end is None:
480            return np.nan

483        if a.is_reverse == b.is_reverse:

Returns np.nan if the pair can't yield a meaningful length.

39    """Check if an alignment overlaps a region. Returns False if not,
40    and the name of the first overlapping region if it does.
